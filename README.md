# even and odd number
num = int(input("enter the value: "))
if num % 2 == 0 
    print(f"{num} is an even number")
else:
    print(f"{num} is an odd number")

# sum 1 to 50 
num1 ,num2 = 1, 50
sum = 0
for i in range(num1 ,num2+1):
    sum += i
print(f"The sum of numbers 1 to 50: {sum}")
